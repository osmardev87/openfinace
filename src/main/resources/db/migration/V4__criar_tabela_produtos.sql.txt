-- Tabela de Produtos / Estoque
CREATE TABLE IF NOT EXISTS produtos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nome VARCHAR(150) NOT NULL,
    descricao TEXT,
    sku VARCHAR(50) UNIQUE,               -- Código do produto
    quantidade INTEGER NOT NULL DEFAULT 0 CHECK (quantidade >= 0),
    quantidade_minima INTEGER NOT NULL DEFAULT 5 CHECK (quantidade_minima >= 0),
    valor_custo DECIMAL(12,2) NOT NULL DEFAULT 0,
    valor_venda DECIMAL(12,2) NOT NULL DEFAULT 0,
    ativo BOOLEAN NOT NULL DEFAULT TRUE,
    
    user_id UUID NOT NULL,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP
);

-- Tabela de Entrada da Nota Fiscal
CREATE TABLE IF NOT EXISTS entrada_nota (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    chave_acesso VARCHAR(44) UNIQUE,  -- ✅ 44 dígitos
    numero_nota VARCHAR(20),
    serie VARCHAR(10),
    data_emissao DATE,
    data_entrada TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fornecedor_nome VARCHAR(150),
    fornecedor_cpf_cnpj VARCHAR(20),
    valor_total DECIMAL(12,2),
    
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Tabela dos itens da nota
CREATE TABLE IF NOT EXISTS entrada_item (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    entrada_nota_id UUID NOT NULL REFERENCES entrada_nota(id) ON DELETE CASCADE,
    produto_id UUID REFERENCES produtos(id), -- ← vincula ao seu produto
    descricao_item VARCHAR(200) NOT NULL,
    quantidade INTEGER NOT NULL CHECK (quantidade > 0),
    valor_unitario DECIMAL(12,2) NOT NULL,
    valor_total DECIMAL(12,2) NOT NULL,
    codigo_fornecedor VARCHAR(50) -- SKU do fornecedor para casar
);
